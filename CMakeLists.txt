cmake_minimum_required(VERSION 3.16)
project(porteus_win7_shell VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(X11 REQUIRED)
add_executable(win7shell src/main.cpp)
target_include_directories(win7shell PRIVATE ${X11_INCLUDE_DIR})
target_link_libraries(win7shell PRIVATE ${X11_LIBRARIES})
target_compile_options(win7shell PRIVATE -Wall -Wextra -Wpedantic)
